const R = {
  recognised: 39607.8,
  cogsVar: 25737.54,
  cogsFixed: 3952.0,
  taxRate: 0.1,
  cash: 31543.39,
  paidBookings: 70,
  newBookings: 71,
  refundsOut: 1826.0,
  refundPayments: 6,
  within: 4,
  after: 2,
  reviews: 10,
  departures: 15,
  status: [
    { key: 'paid', label: 'Paid', n: 65 },
    { key: 'cancelled', label: 'Cancelled', n: 6 },
  ],
};
R.cogs = R.cogsVar + R.cogsFixed;
R.gross = R.recognised - R.cogs;
R.tax = Math.round(R.gross * R.taxRate * 100) / 100;
R.fees = Math.round((R.cash * 0.029 + 0.3 * R.paidBookings) * 100) / 100;
R.net = Math.round((R.gross - R.tax - R.fees) * 100) / 100;
R.margin = R.gross / R.recognised;

const usd = (v) => v.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
const acc = (v) => (v < 0 ? `(${usd(-v)})` : usd(v));
const short = (v) => {
  const a = Math.abs(v);
  const s = a >= 1000 ? `$${(a / 1000).toFixed(1)}k` : `$${Math.round(a)}`;
  return v < 0 ? `−${s}` : s;
};

// ── Thác nước P&L (SVG tĩnh) ────────────────────────────────────────────────
function waterfall() {
  const steps = [
    { label: 'Revenue recognised', v: R.recognised, kind: 'total' },
    { label: 'Cost per traveller', v: -R.cogsVar, kind: 'down' },
    { label: 'Cost per departure', v: -R.cogsFixed, kind: 'down' },
    { label: 'Gross profit', v: R.gross, kind: 'total' },
    { label: 'Tax on margin', v: -R.tax, kind: 'down' },
    { label: 'Payment fees', v: -R.fees, kind: 'down' },
    { label: 'Net profit', v: R.net, kind: 'total' },
  ];
  const W = 480, H = 236, top = 20, bottom = 42, left = 6, right = 6;
  const plotH = H - top - bottom;
  const max = R.recognised;
  const y = (v) => top + plotH - (v / max) * plotH;
  const band = (W - left - right) / steps.length;
  const barW = Math.min(46, band * 0.62);
  let running = 0;
  let out = '';
  // lưới ngang mờ: 0, 20k, 40k
  for (const g of [0, 20000, 40000]) {
    if (g > max * 1.02) continue;
    out += `<line x1="${left}" x2="${W - right}" y1="${y(g)}" y2="${y(g)}" class="grid${g === 0 ? ' base' : ''}"/>`;
    out += `<text x="${W - right}" y="${y(g) - 3}" class="axis" text-anchor="end">${g === 0 ? '$0' : `$${g / 1000}k`}</text>`;
  }
  steps.forEach((s, i) => {
    const cx = left + band * i + band / 2;
    const x = cx - barW / 2;
    let from, to;
    if (s.kind === 'total') { from = 0; to = s.v; running = s.v; }
    else { from = running + s.v; to = running; running = from; }
    const yTop = y(to), yBot = y(from);
    const h = Math.max(1.5, yBot - yTop);
    const fill = s.kind === 'total' ? TEAL : ORANGE;
    const r = 4;
    // đầu dữ liệu bo 4px, chân vuông ở baseline (khối tổng) / ở đáy đoạn (khối trừ)
    const d = s.kind === 'total'
      ? `M${x},${yBot} L${x},${yTop + r} Q${x},${yTop} ${x + r},${yTop} L${x + barW - r},${yTop} Q${x + barW},${yTop} ${x + barW},${yTop + r} L${x + barW},${yBot} Z`
      : `M${x},${yTop} L${x + barW},${yTop} L${x + barW},${yBot - Math.min(r, h / 2)} Q${x + barW},${yBot} ${x + barW - Math.min(r, h / 2)},${yBot} L${x + Math.min(r, h / 2)},${yBot} Q${x},${yBot} ${x},${yBot - Math.min(r, h / 2)} Z`;
    out += `<path d="${d}" fill="${fill}"${s.kind === 'down' ? ' class="hatch-host"' : ''}/>`;
    if (s.kind === 'down') out += `<path d="${d}" fill="url(#hatch)"/>`;
    // nối bậc thang tới cột kế
    if (i < steps.length - 1) {
      const nx = left + band * (i + 1) + band / 2 - barW / 2;
      const ly = s.kind === 'total' ? yTop : yBot;
      out += `<line x1="${x + barW}" x2="${nx}" y1="${ly}" y2="${ly}" class="connector"/>`;
    }
    // nhãn giá trị ở đầu cột (tổng: trên; trừ: dưới đáy đoạn)
    const vy = s.kind === 'total' ? yTop - 5 : yBot + 11;
    out += `<text x="${cx}" y="${vy}" class="val" text-anchor="middle">${short(s.v)}</text>`;
    // nhãn hạng mục dưới trục, hai dòng
    const words = s.label.split(' ');
    const mid = Math.ceil(words.length / 2);
    out += `<text x="${cx}" y="${H - bottom + 15}" class="cat" text-anchor="middle">${words.slice(0, mid).join(' ')}</text>`;
    out += `<text x="${cx}" y="${H - bottom + 26}" class="cat" text-anchor="middle">${words.slice(mid).join(' ')}</text>`;
  });
  const defs = `<defs><pattern id="hatch" width="5" height="5" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="5" stroke="#fff" stroke-opacity=".55" stroke-width="1.6"/></pattern></defs>`;
  return `<svg class="wf" viewBox="0 0 ${W} ${H}" role="img" aria-label="Profit and loss waterfall, September 2026">${defs}${out}</svg>`;
}

// ── Thanh 100% cơ cấu trạng thái ────────────────────────────────────────────
function statusBar() {
  const total = R.status.reduce((a, s) => a + s.n, 0);
  let x = 0;
  let segs = '';
  const W = 600, H = 16, gap = 2;
  R.status.forEach((s, i) => {
    const w = (s.n / total) * W - (i < R.status.length - 1 ? gap : 0);
    const fill = s.key === 'paid' ? TEAL : ORANGE;
    segs += `<rect x="${x}" y="0" width="${w}" height="${H}" rx="3" fill="${fill}"/>`;
    if (s.key !== 'paid') segs += `<rect x="${x}" y="0" width="${w}" height="${H}" rx="3" fill="url(#hatch2)"/>`;
    x += w + gap;
  });
  const defs = `<defs><pattern id="hatch2" width="5" height="5" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="5" stroke="#fff" stroke-opacity=".55" stroke-width="1.6"/></pattern></defs>`;
  const legend = R.status
    .map((s) => `<li><i class="sw ${s.key}"></i><b>${s.label}</b> ${s.n} <span>(${Math.round((s.n / total) * 100)}%)</span></li>`)
    .join('');
  return `<svg class="sb" viewBox="0 0 ${W} ${H}" preserveAspectRatio="none" aria-hidden="true">${defs}${segs}</svg><ul class="legend">${legend}<li class="tot">Total <b>${total}</b></li></ul>`;
}
